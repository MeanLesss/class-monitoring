# Classroom Occupancy - dependencies.
#
# This file is used BOTH locally and by Streamlit Community Cloud (the cloud
# REQUIRES it to be named exactly requirements.txt at the repo root).
#
# We install ultralytics-opencv-headless (the official headless build of
# ultralytics) instead of plain ultralytics. The plain one forces the
# non-headless `opencv-python`, which needs system libs (libGL, libglib2.0-0)
# that Streamlit Community Cloud's Debian Bullseye image cannot provide
# (libglib2.0-0 -> libffi7 missing). The headless build uses
# `opencv-python-headless` and needs NO graphics system libraries, so it runs
# cleanly on the free cloud tier (CPU-only, ~1GB RAM).
#
# streamlit-webrtc + av (below) enable the live WebRTC feed on the cloud too,
# but WebRTC may be unreliable there (needs a TURN relay) - the photo / upload
# paths in cloud_app.py always work regardless.

# Official ultralytics headless build (imports the same `ultralytics` package
# but requires opencv-python-headless).
ultralytics-opencv-headless>=8.4.0
opencv-python-headless>=4.8.0
numpy>=1.24
pandas>=2.0
matplotlib>=3.7
streamlit>=1.61

# Live WebRTC feed (cloud_app.py has a guarded fallback: if these fail to
# install, the photo / upload paths still work).
streamlit-webrtc>=0.47.0
av>=11.0.0
